javascript
// src/utils/result.js

/**
 * @template T
 * @template E
 * @typedef {Object} Success
 * @property {true} success
 * @property {T} value
 */

/**
 * @template T
 * @template E
 * @typedef {Object} Failure
 * @property {false} success
 * @property {E} error
 */

/**
 * @template T
 * @template E
 * @typedef {Success<T, E> | Failure<T, E>} Result
 */

/**
 * Создаёт успешный результат
 * @template T
 * @param {T} value
 * @returns {Success<T, never>}
 */
export function success(value) {
  return { success: true, value };
}

/**
 * Создаёт результат с ошибкой
 * @template E
 * @param {E} error
 * @returns {Failure<never, E>}
 */
export function failure(error) {
  return { success: false, error };
}

/**
 * Проверяет, является ли результат успешным
 * @template T, E
 * @param {Result<T, E>} result
 * @returns {result is Success<T, E>}
 */
export function isSuccess(result) {
  return result.success === true;
}

/**
 * Проверяет, является ли результат ошибочным
 * @template T, E
 * @param {Result<T, E>} result
 * @returns {result is Failure<T, E>}
 */
export function isFailure(result) {
  return result.success === false;
}

// src/calculatorEngine.js

import { success, failure } from './utils/result.js';
import { CalculatorError } from './constants.js';

export const CalculatorEngine = {
  /**
   * @param {number} a
   * @param {number} b
   * @returns {number}
   */
  add(a, b) {
    return a + b;
  },

  /**
   * @param {number} a
   * @param {number} b
   * @returns {number}
   */
  subtract(a, b) {
    return a - b;
  },

  /**
   * @param {number} a
   * @param {number} b
   * @returns {number}
   */
  multiply(a, b) {
    return a * b;
  },

  /**
   * @param {number} a
   * @param {number} b
   * @returns {import('./utils/result.js').Result<number, string>}
   */
  divide(a, b) {
    if (b === 0) {
      return failure(CalculatorError.DIVISION_BY_ZERO);
    }
    return success(a / b);
  },
};

3.1 Создать модуль InputValidator
Файл: src/inputValidator.js

Реализуйте функции валидации чисел и операторов.

javascript
// src/inputValidator.js

import { success, failure } from './utils/result.js';
import { Operator, CalculatorError } from './constants.js';

export const InputValidator = {
  /**
   * Проверяет, является ли строка корректным числом (целым, десятичным, отрицательным)
   * @param {string} input
   * @returns {import('./utils/result.js').Result<number, string>}
   */
  validateNumber(input) {
    // Удаляем пробелы по краям
    const trimmed = input.trim();
    if (trimmed === '') {
      return failure(CalculatorError.INVALID_INPUT);
    }
    const num = Number(trimmed);
    // Проверяем, что получилось конечное число (не NaN, не Infinity)
    if (isNaN(num) || !isFinite(num)) {
      return failure(CalculatorError.INVALID_INPUT);
    }
    return success(num);
  },

  /**
   * Проверяет, является ли строка одним из поддерживаемых операторов
   * @param {string} input
   * @returns {import('./utils/result.js').Result<string, string>}
   */
  validateOperator(input) {
    const trimmed = input.trim();
    if (Object.values(Operator).includes(trimmed)) {
      return success(trimmed);
    }
    return failure(CalculatorError.INVALID_OPERATOR);
  },
};

javascript
// src/resultFormatter.js

import { CalculatorError } from './constants.js';

export const ResultFormatter = {
  /**
   * Форматирует число для отображения.
   * Целые числа выводятся без десятичной точки.
   * Десятичные числа округляются до 10 знаков после запятой.
   * @param {number} value
   * @returns {string}
   */
  formatNumber(value) {
    // Проверяем, является ли число целым (в пределах точности)
    if (Number.isInteger(value)) {
      return value.toString();
    }
    // Ограничиваем до 10 знаков после запятой, убирая лишние нули
    return value.toFixed(10).replace(/\.?0+$/, '');
  },

  /**
   * Преобразует ошибку в понятное пользователю сообщение.
   * @param {Error|string} error
   * @returns {string}
   */
  formatError(error) {
    if (typeof error === 'string') {
      return error;
    }
    if (error instanceof Error) {
      return error.message;
    }
    return String(error);
  },
};

Задача 6  (Выполнить)

Реализация контроллера калькулятора
6.1 Создать модуль CalculatorController
Файл: src/calculatorController.js

Контроллер управляет состоянием калькулятора, координирует валидатор, движок и форматтер.

javascript
// src/calculatorController.js

import { CalculatorEngine } from './calculatorEngine.js';
import { InputValidator } from './inputValidator.js';
import { ResultFormatter } from './resultFormatter.js';
import { isSuccess, isFailure } from './utils/result.js';
import { Operator } from './constants.js';

/**
 * @typedef {Object} CalculatorState
 * @property {number|null} currentValue
 * @property {number|null} previousValue
 * @property {string|null} operator
 * @property {string} displayValue
 * @property {boolean} waitingForOperand
 */

export class CalculatorController {
  constructor() {
    this.state = this.getInitialState();
  }

  /**
   * @private
   * @returns {CalculatorState}
   */
  getInitialState() {
    return {
      currentValue: null,
      previousValue: null,
      operator: null,
      displayValue: '0',
      waitingForOperand: true,
    };
  }

  /**
   * Обрабатывает ввод пользователя (цифры, операторы, команды)
   * @param {string} input
   */
  processInput(input) {
    // Если ввод - оператор
    const opResult = InputValidator.validateOperator(input);
    if (isSuccess(opResult)) {
      this.handleOperator(opResult.value);
      return;
    }

    // Если ввод - число
    const numResult = InputValidator.validateNumber(input);
    if (isSuccess(numResult)) {
      this.handleNumber(numResult.value);
      return;
    }

    // Если ввод - специальная команда (например, "C" или "=")
    if (input === 'C') {
      this.clear();
      return;
    }
    if (input === '=') {
      this.calculate();
      return;
    }

    // Неизвестный ввод
    this.state.displayValue = ResultFormatter.formatError('Неизвестная команда');
  }

  /**
   * @private
   * @param {number} num
   */
  handleNumber(num) {
    if (this.state.waitingForOperand) {
      this.state.currentValue = num;
      this.state.waitingForOperand = false;
    } else {
      // Если уже есть текущее значение, конкатенируем цифры? 
      // В простом калькуляторе обычно каждое нажатие цифры заменяет текущее значение.
      // Здесь мы будем считать, что ввод числа всегда задаёт новое значение.
      // Для простоты: если не ждём операнд, значит текущее значение уже установлено,
      // и новое число должно начать новое вычисление? 
      // По требованиям: "КОГДА пользователь вводит новое число после получения результата, 
      // Калькулятор ДОЛЖЕН начать новое вычисление". Поэтому проверяем:
      if (this.state.previousValue !== null && this.state.operator === null) {
        // Это случай, когда только что получили результат (previousValue хранит результат, operator null)
        this.clear(); // начинаем новое вычисление
        this.state.currentValue = num;
        this.state.waitingForOperand = false;
      } else {
        // Обычный ввод цифр: заменяем текущее значение
        this.state.currentValue = num;
      }
    }
    this.updateDisplayFromCurrent();
  }

  /**
   * @private
   * @param {string} op
   */
  handleOperator(op) {
    // Если уже есть оператор и текущее значение, выполняем предыдущую операцию
    if (this.state.operator !== null && this.state.currentValue !== null && !this.state.waitingForOperand) {
      this.calculate();
    }
    // Сохраняем оператор и предыдущее значение
    this.state.operator = op;
    if (this.state.currentValue !== null) {
      this.state.previousValue = this.state.currentValue;
      this.state.currentValue = null;
      this.state.waitingForOperand = true;
    }
  }

  /**
   * Выполняет вычисление на основе текущего состояния
   */
  calculate() {
    if (this.state.operator === null || this.state.previousValue === null || this.state.currentValue === null) {
      // Недостаточно операндов — ничего не делаем
      return;
    }

    let result;
    const a = this.state.previousValue;
    const b = this.state.currentValue;

    switch (this.state.operator) {
      case Operator.ADD:
        result = CalculatorEngine.add(a, b);
        break;
      case Operator.SUBTRACT:
        result = CalculatorEngine.subtract(a, b);
        break;
      case Operator.MULTIPLY:
        result = CalculatorEngine.multiply(a, b);
        break;
      case Operator.DIVIDE: {
        const divResult = CalculatorEngine.divide(a, b);
        if (isFailure(divResult)) {
          this.state.displayValue = ResultFormatter.formatError(divResult.error);
          // Сохраняем состояние (предыдущее значение и оператор не сбрасываем? По требованию: "сохранить предыдущее корректное состояние")
          // Здесь мы не изменяем состояние, просто показываем ошибку.
          return;
        }
        result = divResult.value;
        break;
      }
      default:
        this.state.displayValue = ResultFormatter.formatError('Неизвестный оператор');
        return;
    }

    // Успешное вычисление
    this.state.previousValue = result;
    this.state.currentValue = null;
    this.state.operator = null;
    this.state.waitingForOperand = true;
    this.state.displayValue = ResultFormatter.formatNumber(result);
  }

  /**
   * Сбрасывает состояние калькулятора
   */
  clear() {
    this.state = this.getInitialState();
  }

  /**
   * Возвращает строку для отображения на дисплее
   * @returns {string}
   */
  getCurrentDisplay() {
    return this.state.displayValue;
  }

  /**
   * @private
   */
  updateDisplayFromCurrent() {
    if (this.state.currentValue !== null) {
      this.state.displayValue = ResultFormatter.formatNumber(this.state.currentValue);
    }
  }
}

import { Operator } from '../../src/constants.js';

describe('Feature: calculator, Property 8: Цепочка операций', () => {
  test('Последовательность операций вычисляется слева направо, результат предыдущей становится первым операндом', () => {
    fc.assert(
      fc.property(
        fc.integer(),
        fc.integer(),
        fc.integer(),
        fc.constantFrom(...Object.values(Operator)),
        fc.constantFrom(...Object.values(Operator)),
        (a, b, c, op1, op2) => {
          // Пропускаем деление на ноль во втором операнде для простоты теста
          fc.pre(!(op1 === Operator.DIVIDE && b === 0));
          fc.pre(!(op2 === Operator.DIVIDE && c === 0));

          const controller = new CalculatorController();
          
          // Вводим a
          controller.processInput(a.toString());
          // Вводим оператор op1
          controller.processInput(op1);
          // Вводим b
          controller.processInput(b.toString());
          // Вычисляем первый шаг (обычно равно не требуется, но для простоты нажмём равно)
          controller.calculate(); // теперь результат a op1 b должен быть в previousValue

          // Затем вводим оператор op2
          controller.processInput(op2);
          // Вводим c
          controller.processInput(c.toString());
          // Вычисляем
          controller.calculate();

          // Ручное вычисление ожидаемого результата
          let firstResult;
          switch (op1) {
            case Operator.ADD: firstResult = a + b; break;
            case Operator.SUBTRACT: firstResult = a - b; break;
            case Operator.MULTIPLY: firstResult = a * b; break;
            case Operator.DIVIDE: firstResult = a / b; break;
          }
          let expected;
          switch (op2) {
            case Operator.ADD: expected = firstResult + c; break;
            case Operator.SUBTRACT: expected = firstResult - c; break;
            case Operator.MULTIPLY: expected = firstResult * c; break;
            case Operator.DIVIDE: expected = firstResult / c; break;
          }

          const display = controller.getCurrentDisplay();
          // Сравниваем с учётом форматирования
          expect(parseFloat(display)).toBeCloseTo(expected, 10);
        }
      ),
      { numRuns: 100 }
    );
  });
